1.3 IAM Policies — Production Team

# Create production-team group
oci iam group create --name "ais-prod-admins" \
  --description "Production admin team · full access in compartment-prod"

# Policy: prod-admins can manage all-resources in compartment-prod
oci iam policy create \
  --compartment-id $TENANCY_OCID \
  --name "prod-admins-policy" \
  --description "Prod admin team permissions" \
  --statements '[
    "Allow group ais-prod-admins to manage all-resources in compartment compartment-prod",
    "Allow group ais-prod-admins to read audit-events in tenancy",
    "Allow group ais-prod-admins to inspect compartments in tenancy"
  ]'
Least Privilege สำหรับ Prod

Prod team ใช้ "manage" เฉพาะใน compartment-prod · ไม่ให้ tenancy-level manage permission · audit events read-only · ใช้ Bastion service สำหรับ SSH access แทน static keys

Compartment + VCN (Multi-AD)

VCN 10.20.0.0/16 + 4 subnets spread across 2 ADs · NSGs · Service Gateway

เป้าหมาย
Compartment "prod" + VCN with Multi-AD subnets · NSGs · Service Gateway · ready for OKE/DB/Cache

2.1 Create Compartment

oci iam compartment create \
  --compartment-id $TENANCY_OCID \
  --name $COMPARTMENT_NAME \
  --description "Production environment for AIS MMT-RFP" \
  --wait-for-state ACTIVE

export COMPARTMENT_OCID=$(oci iam compartment list \
  --compartment-id $TENANCY_OCID \
  --name $COMPARTMENT_NAME \
  --query "data[0].id" --raw-output)

2.2 VCN with Multi-AD Subnets

SubnetCIDRADPurpose
public-lb10.20.0.0/24regionalLoad Balancer (public)
oke-nodes10.20.1.0/22regionalOKE workers across AD-1 + AD-2
oke-pods10.20.8.0/21regionalPod IPs (VCN-native)
private-data10.20.16.0/24regionalDB · Cache · Streaming endpoints
OCI Subnets are Regional

OCI subnets are regional (not AD-bound) by default · resources within a regional subnet can be placed in any AD · ใช้ regional subnets เพื่อ Multi-AD HA · ใน Prod เราจะ deploy DB primary AD-1 + standby AD-2 ในระบบ subnet เดียวกัน

export VCN_OCID=$(oci network vcn create \
  --compartment-id $COMPARTMENT_OCID \
  --display-name $VCN_NAME \
  --cidr-blocks '["10.20.0.0/16"]' \
  --dns-label "vcnprod${SUFFIX}" \
  --wait-for-state AVAILABLE \
  --query "data.id" --raw-output)

# Gateways
oci network internet-gateway create -c $COMPARTMENT_OCID --vcn-id $VCN_OCID --is-enabled true --display-name "igw-prod" --wait-for-state AVAILABLE
oci network nat-gateway create -c $COMPARTMENT_OCID --vcn-id $VCN_OCID --display-name "nat-prod" --wait-for-state AVAILABLE
oci network service-gateway create -c $COMPARTMENT_OCID --vcn-id $VCN_OCID --display-name "sg-prod" \
  --services "[{\"serviceId\":\"$(oci network service list --query 'data[?contains(\"cidr-block\",`all-`)].id|[0]' --raw-output)\"}]" \
  --wait-for-state AVAILABLE

2.3 Network Security Groups (NSGs)

# NSG for OKE workers
oci network nsg create -c $COMPARTMENT_OCID --vcn-id $VCN_OCID --display-name "nsg-oke-workers" --wait-for-state AVAILABLE

# NSG for DB (only OKE workers can access)
oci network nsg create -c $COMPARTMENT_OCID --vcn-id $VCN_OCID --display-name "nsg-db" --wait-for-state AVAILABLE

# NSG for Cache
oci network nsg create -c $COMPARTMENT_OCID --vcn-id $VCN_OCID --display-name "nsg-cache" --wait-for-state AVAILABLE
NSG > Security List

OCI Network Security Groups (NSG) ดีกว่า security list เพราะ: (1) ผูกกับ vNIC เฉพาะ · (2) ไม่ผูกกับ subnet · (3) สามารถ reference NSG อื่นเป็น source (เช่น "DB NSG allows traffic from OKE NSG only") · Prod แนะนำใช้ NSG

3.2 WAF Prevention Mode + Bot Management

Prevention Mode = Block

WAF จะ block requests ที่ match OWASP rules · ทดสอบใน Detection mode ก่อน · ดู false positives · whitelist ก่อนเปลี่ยนเป็น Prevention

cat > waf-prod.json <<'EOF'
{
  "actions": [
    {"name": "Block", "type": "RETURN_HTTP_RESPONSE", "code": 403},
    {"name": "Challenge", "type": "CHECK"}
  ],
  "requestProtection": {
    "rules": [
      {"key": "owasp-crs", "name": "OWASP CRS Top 10",
       "type": "PROTECTION", "actionName": "Block",
       "protectionCapabilities": [
         {"key": "920170", "version": 1},
         {"key": "942100", "version": 1},
         {"key": "941100", "version": 1}
       ]}
    ]
  },
  "requestRateLimiting": {
    "rules": [
      {"name": "rate-limit-per-ip", "actionName": "Block",
       "configurations": [{"period": 60, "requestsLimit": 1000, "actionDuration": 600}]}
    ]
  }
}
EOF

OKE Cluster — Multi-AD HA

2 OCPU/16 GB workers · 2-10 nodes auto-scale · spread across AD-1 + AD-2

4.1 Create OKE Cluster (Production)

ใช้เวลา ~15 นาที

Multi-AD OKE provisioning · grab coffee

export OKE_OCID=$(oci ce cluster create \
  --compartment-id $COMPARTMENT_OCID \
  --name $OKE_NAME \
  --vcn-id $VCN_OCID \
  --kubernetes-version "v1.29.1" \
  --service-lb-subnet-ids "[\"$PUBLIC_SUBNET_OCID\"]" \
  --endpoint-public-ip-enabled true \
  --cluster-pod-network-options '[{"cniType":"OCI_VCN_IP_NATIVE"}]' \
  --wait-for-state SUCCEEDED \
  --query "data.id" --raw-output)

4.2 Multi-AD Node Pool

export AD1=$(oci iam availability-domain list -c $TENANCY_OCID --query "data[0].name" --raw-output)
export AD2=$(oci iam availability-domain list -c $TENANCY_OCID --query "data[1].name" --raw-output)

oci ce node-pool create \
  --compartment-id $COMPARTMENT_OCID \
  --cluster-id $OKE_OCID \
  --name "pool-prod-multiAD" \
  --kubernetes-version "v1.29.1" \
  --node-shape "VM.Standard.E4.Flex" \
  --node-shape-config '{"ocpus":2,"memoryInGBs":16}' \
  --size 4 \
  --placement-configs "[
    {\"availabilityDomain\":\"$AD1\",\"subnetId\":\"$NODE_SUBNET_OCID\"},
    {\"availabilityDomain\":\"$AD2\",\"subnetId\":\"$NODE_SUBNET_OCID\"}
  ]" \
  --ssh-public-key "$SSH_PUB_KEY" \
  --wait-for-state SUCCEEDED
Placement Configs · Multi-AD Spread

กำหนด placement-configs หลาย ADs · OKE จะ distribute nodes evenly · size=4 จะได้ 2 nodes ใน AD-1 + 2 ใน AD-2 · ถ้า AD ล่ม · pods จะ reschedule ไป AD ที่เหลือ

10.2 Deploy with Production-grade Patterns

cat <<EOF | kubectl apply -n queue-prod -f -
apiVersion: apps/v1
kind: Deployment
metadata:
  name: queue-api
spec:
  replicas: 3
  selector: {matchLabels: {app: queue-api}}
  template:
    metadata:
      labels: {app: queue-api, env: prod}
    spec:
      affinity:
        podAntiAffinity:
          preferredDuringSchedulingIgnoredDuringExecution:
          - weight: 100
            podAffinityTerm:
              labelSelector: {matchLabels: {app: queue-api}}
              topologyKey: kubernetes.io/hostname
      topologySpreadConstraints:
      - maxSkew: 1
        topologyKey: oci.oraclecloud.com/fault-domain
        whenUnsatisfiable: DoNotSchedule
        labelSelector: {matchLabels: {app: queue-api}}
      containers:
      - name: api
        image: ${REGION}.ocir.io/${OS_NAMESPACE}/queue-api:1.0.0
        ports: [{containerPort: 8080}]
        resources:
          requests: {cpu: 200m, memory: 256Mi}
          limits:   {cpu: 1000m, memory: 1Gi}
        readinessProbe:
          httpGet: {path: /healthz, port: 8080}
          initialDelaySeconds: 5
          periodSeconds: 10
        livenessProbe:
          httpGet: {path: /healthz, port: 8080}
          initialDelaySeconds: 30
          periodSeconds: 30
---
apiVersion: policy/v1
kind: PodDisruptionBudget
metadata:
  name: queue-api-pdb
spec:
  minAvailable: 2
  selector: {matchLabels: {app: queue-api}}
---
apiVersion: autoscaling/v2
kind: HorizontalPodAutoscaler
metadata:
  name: queue-api-hpa
spec:
  scaleTargetRef:
    apiVersion: apps/v1
    kind: Deployment
    name: queue-api
  minReplicas: 3
  maxReplicas: 50
  metrics:
  - type: Resource
    resource: {name: cpu, target: {type: Utilization, averageUtilization: 60}}
EOF
3 Production Pod Patterns

(1) PodAntiAffinity · pods spread across nodes · (2) TopologySpreadConstraints · pods spread across ADs · (3) PodDisruptionBudget minAvailable=2 · ห้าม drain pods ที่จะให้เหลือ < 2
